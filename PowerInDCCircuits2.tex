from pylab import *
import random

from randassign import RandAssign
ra = RandAssign()

def make_problem():
	def rand_choice1():
		return random.choice([40,60,75,100])
	def rand_choice2():
		return random.choice([15,20])
	
	choice1 = rand_choice1()
	choice2 = rand_choice2()
	
	problem_template = r'''
	Before the availability of high-efficiency compact fluorescent and LED bulbs, most light bulbs in homes were incandescent light bulbs. Treating the incandescent bulbs as a resistive element, what is the maximum number of $\SI{{{0:.2g}}}{{\watt}}$ light bulbs you were able to connect in parallel in a 120-V home circuit without tripping the $\SI{{{1:.2g}}}{{\ampere}}$ circuit breaker?
	'''
	
	P = 120*choice2
	num = P / choice1
	
	problem = problem_template.format(choice1,choice2)
	return(problem,num)
	
p, num = make_problem()

t = '\\SI{{{0:.3g}}} light bulbs'
ra.addsoln(t.format(num))

print(p)
