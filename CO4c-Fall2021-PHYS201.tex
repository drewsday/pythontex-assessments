% The whole assessment is generated from the single pycode block below.
% Layout is COPY-MAJOR: the outer loop walks over copies of the assessment and
% the inner loop walks over the four questions, so each copy comes out as a
% complete paper (Q1-Q4) rather than as a run of same-numbered pages.

\begin{pycode}
import math
import random

from randassign import RandAssign
ra = RandAssign()

# Only the standard-library RNG is used here.  An earlier version of this file
# did `from pylab import *`, which shadowed `uniform` with numpy.random.uniform
# while `randrange` stayed on the stdlib generator; random.seed() then drove only
# one of the two streams and the papers could not be regenerated.  Seeding now
# controls every draw in this document.
#
# Leave SEED as None for a fresh set of papers on each build.  Set it to an
# integer to reproduce an exact set of papers (e.g. to reprint a lost key).
SEED = None
if SEED is not None:
    random.seed(SEED)

NCOPIES = 4          # number of distinct assessments to generate
G = 9.81             # m/s^2


def rand_mass():
    return round(random.uniform(1, 3), 2)


def rand_stretch():
    return round(random.uniform(0.05, 0.39), 2)


def rand_accel():
    return round(random.uniform(4, 8), 2)


def rand_k():
    return random.randrange(19, 51)


def rand_angle():
    return random.randrange(25, 40)


def q_spring_constant():
    """Horizontal, frictionless: k = m a / x.  Answer is a spring constant."""
    mass = rand_mass()
    stretch = rand_stretch()
    accel = rand_accel()

    k = mass * accel / stretch

    text = (
        r'An object attached to an ideal massless spring is pulled across a '
        r'frictionless surface. If the mass of the object is '
        r'$m = \SI{%.3g}{\kilogram}$ and the spring is stretched by '
        r'$\SI{%.3g}{\meter}$ when the object is accelerating at '
        r'$\SI{%.3g}{\meter\per\second\squared}$, what is the spring constant?'
    ) % (mass, stretch, accel)

    # Unit is newton per metre.  The previous version reported this answer in
    # kilograms because it reused the format string from the question below.
    return text, r'$k = \SI{%.3g}{\newton\per\meter}$' % k


def q_mass_from_k():
    """Horizontal, frictionless: m = k x / a.  Answer is a mass."""
    stretch = rand_stretch()
    accel = rand_accel()
    k = rand_k()

    mass = k * stretch / accel

    text = (
        r'An object attached to an ideal massless spring is pulled across a '
        r'frictionless surface. If the spring constant is '
        r'$k = \SI{%.3g}{\newton\per\meter}$ and the spring is stretched by '
        r'$\SI{%.3g}{\meter}$ when the object is accelerating at '
        r'$\SI{%.3g}{\meter\per\second\squared}$, what is the mass of the object?'
    ) % (k, stretch, accel)

    return text, r'$m = \SI{%.3g}{\kilogram}$' % mass


# Q3 and Q4 use diagram-springs.png: two identical springs hang from the ceiling
# in a V and meet at the block.  Theta is measured from the vertical dashed line,
# so each spring contributes F*cos(theta) vertically and equilibrium gives
#     2 k x cos(theta) = m g.
FIGURE = (
    r'\begin{figure}[h]\centering'
    r'\includegraphics[width=10cm,keepaspectratio]{diagram-springs}'
    r'\end{figure}'
)


def q_stretch_from_mass():
    """x = m g / (2 k cos(theta)).  Answer is a length."""
    while True:
        mass = rand_mass()
        k = rand_k()
        angle = rand_angle()
        stretch = mass * G / (2 * k * math.cos(math.radians(angle)))
        # Keep the stretch in a range that is sane for a real spring.
        if 0.05 <= stretch <= 0.60:
            break

    text = (
        r'Two identical springs with spring constant '
        r'$k = \SI{%.3g}{\newton\per\meter}$ hold a '
        r'$\SI{%.3g}{\kilogram}$ hanging mass as in the figure below. '
        r'Suppose $\theta = \ang{%d}$. How much has one of the springs been '
        r'stretched from its original length?'
    ) % (k, mass, angle)

    return text + FIGURE, r'$x = \SI{%.3g}{\meter}$' % stretch


def q_mass_from_stretch():
    """m = 2 k x cos(theta) / g.  Answer is a mass."""
    while True:
        stretch = rand_stretch()
        k = rand_k()
        angle = rand_angle()
        mass = 2 * k * stretch * math.cos(math.radians(angle)) / G
        # Keep the hanging mass in the same range the other questions use.
        if 0.5 <= mass <= 4.0:
            break

    text = (
        r'Two identical springs with spring constant '
        r'$k = \SI{%.3g}{\newton\per\meter}$ hold a hanging object as in the '
        r'figure below. Suppose $\theta = \ang{%d}$ and that the springs have '
        r'each been stretched $\SI{%.3g}{\meter}$ from their original length. '
        r'What is the mass of the object?'
    ) % (k, angle, stretch)

    return text + FIGURE, r'$m = \SI{%.3g}{\kilogram}$' % mass


QUESTIONS = (
    q_spring_constant,
    q_mass_from_k,
    q_stretch_from_mass,
    q_mass_from_stretch,
)

HEADER = (
    r'\noindent\begin{tabular}{l l r}'
    r'\textbf{\class} - \textbf{\examnum} & '
    r'\textbf{Name} \makebox[3.25in]{\hrulefill}\\'
    r'\end{tabular}\\'
    r'\rule[1ex]{\textwidth}{2.21pt}'
)

seen = set()
copies = 0
while copies < NCOPIES:
    built = [q() for q in QUESTIONS]
    fingerprint = tuple(text for text, _ in built)
    if fingerprint in seen:
        continue          # never print the same paper twice
    seen.add(fingerprint)
    copies += 1

    print(HEADER)
    # Reset explicitly so every copy numbers its questions 1-4, regardless of
    # how many `questions` environments came before it.
    print(r'\setcounter{question}{0}')
    print(r'\begin{questions}')
    for text, _ in built:
        print(r'\question ' + text)
    print(r'\end{questions}')
    print(r'\newpage')

    # One key entry per copy, in the same order the questions were printed,
    # so the answer key lines up paper-for-paper with the generated exams.
    ra.addsoln([soln for _, soln in built])
\end{pycode}
